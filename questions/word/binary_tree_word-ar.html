الجزء هـ: عدّ العقد الزوجية (3 درجات)

public int countEven(Node root) {
    if (root == null) {
        return 0;
    }
    
    int count = 0;
    if (root.data % 2 == 0) {
        count = 1;
    }
    
    return count + countEven(root.left) + countEven(root.right);
}
